const SP = PropertiesService.getScriptProperties(); const CONFIG = { NOTION_API_KEY: SP.getProperty("NOTION_API_KEY"), DB_IDS: { MAIN: SP.getProperty("DB_MAIN"), STUDIO: SP.getProperty("DB_STUDIO"), AUTHOR: SP.getProperty("DB_AUTHOR"), DIRECTOR: SP.getProperty("DB_DIRECTOR"), CHARA_DESIGN: SP.getProperty("DB_CHARA_DESIGN"), MUSIC: SP.getProperty("DB_MUSIC") } }; function doGet(e) { const action = e.parameter.action; const keyword = e.parameter.keyword; try { if (action === "search") { const results = searchWikipediaAnimeRobust(keyword); return createResponse({ status: "success", data: results }); } return createResponse({ status: "error", message: "不明なアクションです" }); } catch (err) { return createResponse({ status: "error", message: err.toString() }); } } function doPost(e) { try { const contents = e.postData.contents; const req = JSON.parse(contents); const action = req.action; if (action === "register") { const result = registerToNotion(req.workData); return createResponse({ status: "success", pageId: result }); } return createResponse({ status: "error", message: "不明なアクションです" }); } catch (err) { return createResponse({ status: "error", message: err.toString() }); } } function createResponse(data) { const output = ContentService.createTextOutput(JSON.stringify(data)); output.setMimeType(ContentService.MimeType.JSON); return output; } // 安定性100% Wikipedia検索 function searchWikipediaAnimeRobust(keyword) { if (!keyword) return []; const cleanKeyword = keyword.trim(); // 1. 記事検索 const searchUrl = `https://ja.wikipedia.org/w/api.php?action=query&list=search&srsearch=${encodeURIComponent(cleanKeyword)}&format=json&utf8=1`; const searchRes = JSON.parse(UrlFetchApp.fetch(searchUrl).getContentText()); if (!searchRes.query || !searchRes.query.search || searchRes.query.search.length === 0) { return []; } const hits = searchRes.query.search; let targetTitle = ""; // アニメ専用記事を最優先で選定 const animeSpecificMatch = hits.find(h => { const t = h.title.toLowerCase(); const k = cleanKeyword.toLowerCase(); return (t === `${k} (アニメ)` || t === `${k}（アニメ）` || (t.includes(k) && t.includes("アニメ"))); }); if (animeSpecificMatch) { targetTitle = animeSpecificMatch.title; } else { const exactMatch = hits.find(h => h.title.toLowerCase() === cleanKeyword.toLowerCase()); if (exactMatch) { targetTitle = exactMatch.title; } else { const partialMatch = hits.find(h => h.title.includes(cleanKeyword)); targetTitle = partialMatch ? partialMatch.title : hits[0].title; } } let pageData = fetchPageWikitext(targetTitle); if (!pageData) return []; // 原作記事だった場合の安全弁（冒頭のアニメ誘導を追跡） const otherusesMatch = pageData.wikitext.match(/\{\{Otheruses[^\}]*?([^\}\|]+アニメ[^\}\|]*)\}\}/i) || pageData.wikitext.match(/アニメ(?:化作品)?については[「『]([^」』]+)[」』]/); if (otherusesMatch && otherusesMatch[1] && otherusesMatch[1] !== targetTitle) { const candidateAnimeTitle = otherusesMatch[1].replace(/\[\[\vert{}\]\]/g, "").trim(); const animePageData = fetchPageWikitext(candidateAnimeTitle); if (animePageData && animePageData.wikitext.includes("アニメ")) { pageData = animePageData; } } return parseWikitextToSeasons(pageData.wikitext, pageData.title, pageData.pageUrl, pageData.mobileUrl, cleanKeyword); } function fetchPageWikitext(title) { const parseUrl = `https://ja.wikipedia.org/w/api.php?action=query&prop=revisions&titles=${encodeURIComponent(title)}&rvslots=*&rvprop=content&redirects=1&format=json&utf8=1`; const parseRes = JSON.parse(UrlFetchApp.fetch(parseUrl).getContentText()); const pages = parseRes.query && parseRes.query.pages ? Object.values(parseRes.query.pages) : []; if (pages.length === 0 || !pages[0].revisions) return null; const pageObj = pages[0]; const resolvedTitle = pageObj.title || title; return { title: resolvedTitle, pageUrl: `https://ja.wikipedia.org/wiki/${encodeURIComponent(resolvedTitle)}`, mobileUrl: `https://ja.m.wikipedia.org/wiki/${encodeURIComponent(resolvedTitle)}`, wikitext: (pageObj.revisions[0].slots && pageObj.revisions[0].slots.main) ? (pageObj.revisions[0].slots.main["*"] || "") : "" }; } function parseWikitextToSeasons(wikitext, title, pageUrl, mobileUrl, keyword) { const clean = (s) => (s || "") .replace(/([\s\S]*?)<\/nowiki>/gi, "$1") .replace(/\[\[[^\]\vert{}]+\Vert{}([^\]]+)\]\]/g, "$1") .replace(/\[\[([^\]]+)\]\]/g, "$1") .replace(/\{\{[^}]+\}\}/g, "") .replace(/]*>[\s\S]*?<\/ref>/gi, "") .replace(//gi, "、") .trim(); const getF = (key) => { const reg = new RegExp(`\\|\\s*${key}\\s*=([^\\n|]+)`, "i"); const m = wikitext.match(reg); return m ? clean(m[1]) : ""; }; const splitNames = (str) => clean(str) .split(/[,、\/\n]/) .map(s => s.replace(/（[^）]*）|\([^)]*\)/g, "").trim()) .filter(Boolean); let genres = []; const gm = wikitext.match(/\|\s*ジャンル\s*=([\s\S]*?)(?=\n\s*[|}])/i); if (gm) { genres = clean(gm[1]) .split(/[,、・\/\n\s*]+/) .map(s => s.trim().replace(/^[\*:\s]+/, "")) .filter(s => s && !s.includes("=") && !s.includes("Infobox") && s !== "アニメ"); } genres = [...new Set(genres)]; const authors = splitNames(getF("原作") || getF("原案") || getF("著者")); const defaultStudios = splitNames(getF("アニメーション制作") || getF("制作")); const defaultDirectors = splitNames(getF("監督") || getF("総監督")); const defaultCharas = splitNames(getF("キャラクターデザイン") || getF("キャラデザ")); const defaultMusics = splitNames(getF("音楽")); let titleRaw = clean(getF("タイトル")) || title.replace(/\s*\(.*?\)$/, ""); const titleParts = titleRaw.split(/\s*\/\s*/).map(t => t.trim()).filter(Boolean); const hasSeason2 = wikitext.includes("放送期間2") || wikitext.includes("放送時間2") || titleParts.length > 1; let baseDisplayTitle = titleRaw; if (keyword && (title.includes("シリーズ") || titleParts.length === 0)) { baseDisplayTitle = keyword; } const results = []; const seasonCount = hasSeason2 ? Math.max(titleParts.length, 2) : 1; for (let sIdx = 0; sIdx < seasonCount; sIdx++) { const seasonNum = sIdx + 1; let sTitle = titleParts[sIdx] || baseDisplayTitle; if (seasonNum >= 2) { if (!sTitle.includes("期")) { sTitle = `${sTitle} (第${seasonNum}期)`; } else { sTitle = sTitle.replace(/（/g, " (").replace(/）/g, ")"); } } else { sTitle = sTitle.replace(/\s*[(（]第1期[)）]/g, "").trim(); sTitle = sTitle.replace(/（/g, " (").replace(/）/g, ")"); } const sStudios = splitNames(getF(`アニメーション制作${seasonNum}`) || getF(`制作${seasonNum}`)) || defaultStudios; const sDirectors = splitNames(getF(`監督${seasonNum}`) || getF(`総監督${seasonNum}`)) || defaultDirectors; const sCharas = splitNames(getF(`キャラクターデザイン${seasonNum}`) || getF(`キャラデザ${seasonNum}`)) || defaultCharas; const sMusics = splitNames(getF(`音楽${seasonNum}`)) || defaultMusics; results.push({ seasonNum: seasonNum, title: sTitle.trim(), wikiUrl: pageUrl, mobileUrl: mobileUrl, genres: genres, studios: sStudios.length > 0 ? sStudios : defaultStudios, authors: authors, directors: sDirectors.length > 0 ? sDirectors : defaultDirectors, charaDesigns: sCharas.length > 0 ? sCharas : defaultCharas, musics: sMusics.length > 0 ? sMusics : defaultMusics }); } return results; } function findOrCreateRelation(dbId, name) { if (!name || name.trim() === "") return null; name = name.trim(); const searchUrl = `https://api.notion.com/v1/databases/${dbId}/query`; const headers = { "Authorization": `Bearer ${CONFIG.NOTION_API_KEY}`, "Notion-Version": "2022-06-28", "Content-Type": "application/json" }; const searchRes = UrlFetchApp.fetch(searchUrl, { method: "post", headers: headers, payload: JSON.stringify({ filter: { property: "名前", title: { equals: name } } }), muteHttpExceptions: true }); const searchData = JSON.parse(searchRes.getContentText()); if (searchData.results && searchData.results.length > 0) { return searchData.results[0].id; } const createUrl = "https://api.notion.com/v1/pages"; const createRes = UrlFetchApp.fetch(createUrl, { method: "post", headers: headers, payload: JSON.stringify({ parent: { database_id: dbId }, properties: { "名前": { title: [{ text: { content: name } }] } } }) }); return JSON.parse(createRes.getContentText()).id; } function registerToNotion(work) { const getRelationIds = (dbId, names) => { return (names || []) .map(n => findOrCreateRelation(dbId, n)) .filter(id => id !== null) .map(id => ({ id: id })); }; const properties = { "作品名": { title: [{ text: { content: work.title } }] }, "制作会社": { relation: getRelationIds(CONFIG.DB_IDS.STUDIO, work.studios) }, "原作者": { relation: getRelationIds(CONFIG.DB_IDS.AUTHOR, work.authors) }, "監督": { relation: getRelationIds(CONFIG.DB_IDS.DIRECTOR, work.directors) }, "キャラデザ": { relation: getRelationIds(CONFIG.DB_IDS.CHARA_DESIGN, work.charaDesigns) }, "音楽": { relation: getRelationIds(CONFIG.DB_IDS.MUSIC, work.musics) }, "放送クール": { multi_select: (work.seasons || []).map(s => ({ name: s })) }, "ジャンル": { multi_select: (work.genres || []).map(g => ({ name: g })) } }; if (work.status) { properties["視聴状況"] = { status: { name: work.status } }; } const payload = { parent: { database_id: CONFIG.DB_IDS.MAIN }, properties: properties }; const res = UrlFetchApp.fetch("https://api.notion.com/v1/pages", { method: "post", headers: { "Authorization": `Bearer ${CONFIG.NOTION_API_KEY}`, "Notion-Version": "2022-06-28", "Content-Type": "application/json" }, payload: JSON.stringify(payload) }); return JSON.parse(res.getContentText()).id; }
